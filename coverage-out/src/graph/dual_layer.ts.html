 *   Groups events/nodes into "topics" with semantic links.

function generateIdWithPrefix(prefix: string): string {
  const hex = "0123456789abcdef";
  let id = "";
  for (let i = 0; i < 12; i++) {
    id += hex[Math.floor(Math.random() * 16)];
  }
  return `${prefix}_${id}`;
}
