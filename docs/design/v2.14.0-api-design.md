## 1. MultiGraphMemory API

```python
class MultiGraphMemory:
    """四正交图索引层"""

    # ── 节点管理 ──

    def add_node(self, memory_id: str, content: str,
                 node_type: NodeType, **metadata) -> None:
        """注册记忆节点到图中

        Args:
            memory_id: 记忆唯一 ID
            content: 记忆内容
            node_type: 节点类型 (EPISODE/FACT/REFLECTION/CONCEPT)
            **metadata: 额外元数据 (session_id, timestamp, ...)

        Idempotent: 多次调用同一 memory_id 不重复添加
        """

    def get_node(self, memory_id: str) -> Optional[Node]:
        """获取节点信息
        Returns: Node 或 None
        """

    def node_count(self) -> int:
        """节点总数"""

    # ── 边管理 ──

    def add_edge(self, source_id: str, target_id: str,
                 edge_type: EdgeType, weight: float = 1.0) -> None:
        """添加边，自动路由到对应子图

        Args:
            source_id: 源节点 ID
            target_id: 目标节点 ID
            edge_type: 边类型
            weight: 初始权重 (默认 1.0)

        子图路由:
            NEXT           → TEMPORAL 子图
            DERIVED_FROM   → CAUSAL 子图
            SYNTHESIZED_FROM→ CAUSAL 子图
            RELATED_TO     → SEMANTIC 子图
            HAS_CONCEPT    → ENTITY 子图
        """

    # ── 检索 ──

    def get_related(self, memory_id: str, subgraph: SubGraphType,
                    limit: int = 10) -> List[str]:
        """获取指定子图中与 memory_id 相关的节点

        Args:
            memory_id: 起始节点
            subgraph: 目标子图
            limit: 最大返回数

        Returns:
            相关节点 ID 列表 (按权重降序)
        """

    def get_expanded_nodes(self, node_ids: List[str],
                           subgraphs: List[SubGraphType] = None,
                           max_depth: int = 1,
                           max_expansion: int = 50) -> Dict[str, float]:
        """从种子节点出发进行图扩展

        Args:
            node_ids: 种子节点 ID 列表
            subgraphs: 要查询的子图 (默认全部)
            max_depth: BFS 最大深度 (默认 1)
            max_expansion: 最大扩展节点数

        Returns:
            {node_id: aggregated_weight}
        """

    def multi_graph_search(self, query_embedding: List[float],
                           sem_nodes: List[str],
                           k: int = 10) -> List[Tuple[str, float]]:
        """多图联合检索

        策略:
          1. 语义种子评分 1.0
          2. 四个子图并行扩展
          3. 加权合并排序

        Args:
            query_embedding: 查询向量
            sem_nodes: 语义检索得到的种子节点 ID
            k: 返回数量

        Returns:
            [(node_id, combined_score), ...] 按分数降序
        """

    # ── 衰减集成 ──

    def apply_decay(self,
                    edge_weights: Dict[Tuple[str, str], float]) -> int:
        """批量更新边权重
        Returns: 成功更新的边数
        """

    def remove_expired_edges(self, threshold: float = 0.05) -> int:
        """移除权重低于阈值的边
        Returns: 被移除的边数
        """

    # ── 统计 ──

    def get_stats(self) -> dict:
        """获取各子图统计信息
        Returns: {'total_nodes': int, 'subgraphs': {...}}
        """

    # ── 持久化 ──

    def to_dict(self) -> dict:
        """序列化为字典"""

    @classmethod
    def from_dict(cls, d: dict) -> "MultiGraphMemory":
        """从字典反序列化"""
```

### 1.1 使用示例

```python
# 创建图
mg = MultiGraphMemory()

# 添加节点
mg.add_node("mem_1", "User prefers Chinese", NodeType.EPISODE,
            session_id="sess_123")
mg.add_node("mem_2", "Building REST API with FastAPI", NodeType.EPISODE,
            session_id="sess_123")

# 添加边 (自动路由)
mg.add_edge("mem_1", "mem_2", EdgeType.NEXT, weight=1.0)
mg.add_edge("mem_1", "mem_3", EdgeType.RELATED_TO, weight=0.85)

# 检索
related = mg.get_related("mem_1", SubGraphType.TEMPORAL, limit=10)
expanded = mg.get_expanded_nodes(["mem_1"], max_depth=2)
results = mg.multi_graph_search(query_emb, ["mem_1"], k=10)

# 衰减
update = {("mem_1", "mem_2"): 0.7}
mg.apply_decay(update)
mg.remove_expired_edges(threshold=0.05)
```

```python
class DualLayerMemory:
    """双层记忆结构"""

    # ── Layer 1: 事件进展图 ──

    def add_event(self, title: str, description: str = "",
                  node_ids: List[str] = None,
                  session_id: str = None,
                  tags: List[str] = None) -> str:
        """创建事件

        Args:
            title: 事件标题
            description: 事件描述
            node_ids: 关联的记忆节点 ID
            session_id: 所属会话 ID
            tags: 标签列表

        Returns:
            event_id (格式: evt_<hex12>)

        自动行为:
          - 与同 session 的最近事件建立时序链
        """

    def link_events(self, event1_id: str, event2_id: str) -> None:
        """显式关联两个事件 (双向)"""

    def get_event_chain(self, event_id: str) -> List[Event]:
        """获取事件链 (前驱事件序列)
        Returns: 按时间排序的事件列表
        """

    def get_event(self, event_id: str) -> Optional[Event]:
        """获取事件详情"""

    def find_events_by_tags(self, tags: List[str]) -> List[Event]:
        """按标签查找事件"""

    def event_count(self) -> int:
        """事件总数"""

    # ── Layer 2: 主题关联网络 ──

    def add_topic(self, name: str, description: str = "",
                  node_ids: List[str] = None,
                  event_ids: List[str] = None,
                  keywords: List[str] = None) -> str:
        """创建主题

        Args:
            name: 主题名称
            description: 描述
            node_ids: 关联节点
            event_ids: 关联事件
            keywords: 关键词标签

        Returns:
            topic_id (格式: tpc_<hex12>)
        """

    def link_topics(self, topic1_id: str, topic2_id: str,
                    weight: float = 0.5) -> None:
        """关联两个主题 (对称)"""

    def get_related_topics(self, topic_id: str,
                           min_weight: float = 0.3) -> List[Tuple[Topic, float]]:
        """获取相关主题
        Returns: [(Topic, weight), ...] 按权重降序
        """

    def search_by_keywords(self, keywords: List[str]) -> List[Topic]:
        """按关键词搜索主题"""

    def topic_count(self) -> int:
        """主题总数"""

    # ── 持久化 ──

    def to_dict(self) -> dict:
        """序列化"""

    @classmethod
    def from_dict(cls, d: dict) -> "DualLayerMemory":
        """反序列化"""
```

## 3. DecayController API

```python
class DecayController:
    """遗忘控制器"""

    def __init__(self, graph: MultiGraphMemory,
                 config: DecayConfig = None):
        """初始化

        Args:
            graph: 关联的图结构
            config: 衰减配置 (默认使用 DecayConfig.default())
        """

    def calculate_weight(self, initial_weight: float,
                         days_elapsed: float,
                         category: str) -> float:
        """计算衰减后权重

        Args:
            initial_weight: 初始权重
            days_elapsed: 经过天数
            category: 半衰期类别 ('temporal', 'semantic', ...)

        Returns:
            衰减后权重 (0.0 ~ 1.0)
        """

    def get_decay_weight(self, edge_tuple: Tuple[str, str],
                         edge_type: str,
                         created_at: float) -> float:
        """计算具体边的衰减权重"""

    def compute_all_decays(self) -> Dict[Tuple[str, str], float]:
        """计算所有边的衰减权重
        Returns: {(source, target): new_weight}
        """

    def classify_edges(self) -> Dict[str, List]:
        """分类所有边
        Returns: {strong: [...], medium: [...], weak: [...], expired: [...]}
        """

    def should_remove_edge(self, source: str, target: str,
                           weight: float) -> bool:
        """判断边是否应被删除"""

    def cleanup_expired(self) -> List[Tuple[str, str]]:
        """清理过期边
        Returns: 被删除的边列表
        """

    def get_stats(self) -> dict:
        """获取衰减统计信息"""
```

### 3.1 使用示例

```python
config = DecayConfig(
    half_life_temporal=7.0,
    half_life_semantic=90.0,
    purge_threshold=0.05,
)

controller = DecayController(multi_graph, config=config)

# 计算单条边权重
w = controller.calculate_weight(1.0, 3.5, "temporal")  # ≈0.7

# 分类所有边
classified = controller.classify_edges()
print(f"Strong: {len(classified['strong'])}")
print(f"Expired: {len(classified['expired'])}")

# 清理
removed = controller.cleanup_expired()
print(f"Removed {len(removed)} edges")
```

## 4. DecayScheduler API

```python
class DecayScheduler:
    """衰减调度器"""

    def __init__(self, controller: DecayController,
                 config: DecayConfig = None):
        """初始化

        调度策略:
          - 定时: 每 decay_interval_hours 小时执行一次
          - 事件: 每 stores_per_decay 次 store() 后执行
          - 手动: schedule_now() 立即执行
        """

    def start(self) -> None:
        """启动定期调度 (后台 daemon 线程)"""

    def stop(self) -> None:
        """停止调度 (等待当前周期完成)"""

    def schedule_now(self) -> None:
        """手动触发立即衰减 (非阻塞)"""

    def notify_store(self) -> None:
        """通知一次 store() 调用发生"""

    def on_complete(self, callback: Callable[[List], None]) -> None:
        """注册衰减完成回调
        Args:
            callback: 接收被删除的边列表
        """

    def is_running(self) -> bool:
        """调度器是否在运行"""

    def get_next_run_time(self) -> Optional[float]:
        """下次定时执行的时间戳"""
```

### 4.1 使用示例

```python
scheduler = DecayScheduler(controller, config)

# 注册完成回调
def on_decay(removed_edges):
    logger.info(f"Decay clean: {len(removed_edges)} edges removed")

scheduler.on_complete(on_decay)

# 启动后台调度
scheduler.start()

# 手动触发
scheduler.schedule_now()

# 每次 store 时通知
scheduler.notify_store()

# 停止
scheduler.stop()
```

## 5. GroundTruthStore API

```python
class GroundTruthStore:
    """真值存储"""

    def __init__(self, workspace: str = None):
        """初始化
        Args:
            workspace: 工作目录 (默认 ~/.claw-mem)
        """

    def store_turn(self, session_id: str, messages: List[Dict],
                   metadata: Dict = None) -> str:
        """存储一轮或多轮对话

        Args:
            session_id: OpenClaw 会话 ID
            messages: [{"role": "user/assistant", "content": "..."}, ...]
            metadata: 额外元数据

        Returns:
            record_id (格式: gt_<hex16>)
        """

    def store_session(self, session_id: str,
                      all_messages: List[Dict],
                      metadata: Dict = None) -> str:
        """存储整个会话 (agent_end 时调用)"""

    def get_session(self, session_id: str) -> List[Dict]:
        """获取指定会话的全部原始对话记录"""

    def search(self, session_id: str = None,
               keyword: str = None,
               limit: int = 50) -> List[Dict]:
        """搜索原始对话

        Args:
            session_id: 限定会话 (可选)
            keyword: 关键词 (可选)
            limit: 最大返回数
        """

    def list_sessions(self) -> List[Dict]:
        """列出所有已存储的会话
        Returns: [{"session_id": ..., "file_size": ..., "last_modified": ...}, ...]
        """

    def count_records(self) -> int:
        """总记录数"""
```

### 5.1 使用示例

```python
gt = GroundTruthStore("~/.claw-mem")

# 存储对话
gt.store_turn(
    session_id="sess_123",
    messages=[
        {"role": "user", "content": "I prefer Chinese"},
        {"role": "assistant", "content": "好的，我会用中文回复。"},
    ],
    metadata={"task": "preference_setting"},
)

# 搜索
results = gt.search(keyword="Chinese")
for r in results:
    print(f"Session: {r['session_id']}, Content: {r['message']['content']}")

# 会话列表
for s in gt.list_sessions():
    print(f"{s['session_id']}: {s['file_size']} bytes")
```

## 7. Bridge RPC 方法

### 7.1 新增 RPC 方法

```python
# bridge.py 中新增 handlers:

"get_graph_stats": self._handle_get_graph_stats,
"get_node_graph": self._handle_get_node_graph,
"persist_graph": self._handle_persist_graph,
"get_decay_stats": self._handle_get_decay_stats,
"force_decay_cycle": self._handle_force_decay_cycle,
"search_ground_truth": self._handle_search_ground_truth,
"list_sessions": self._handle_list_sessions,

def _handle_get_graph_stats(self, params: Dict) -> Dict:
    return self.memory_manager.get_graph_stats()

def _handle_get_node_graph(self, params: Dict) -> Dict:
    return self.memory_manager.get_node_graph(
        params.get("memory_id", "")
    )

def _handle_persist_graph(self, params: Dict) -> Dict:
    return {"success": self.memory_manager.persist_graph()}

def _handle_get_decay_stats(self, params: Dict) -> Dict:
    return self.memory_manager.get_decay_stats()

def _handle_force_decay_cycle(self, params: Dict) -> Dict:
    count = self.memory_manager.force_decay_cycle()
    return {"removed_edges": count}

def _handle_search_ground_truth(self, params: Dict) -> Dict:
    return {
        "results": self.memory_manager.search_ground_truth(
            session_id=params.get("session_id"),
            keyword=params.get("keyword"),
            limit=params.get("limit", 50),
        )
    }

def _handle_list_sessions(self, params: Dict) -> Dict:
    return {"sessions": self.memory_manager.list_sessions()}
```

```typescript
      if (stats?.total_nodes > 0) {
        sections.push(
          `## Memory Graph\n` +
          `- ${stats.total_nodes} nodes, ${stats.total_edges} edges\n` +
          `- Strong edges: ${stats.strong_edges}, Weak: ${stats.weak_edges}`
        );
      }
    } catch { /* 静默失败 */ }
  }

  return sections;
},
```
