## 1. DecoupledRetriever API

统一的解耦检索器，串接 Engram → Spreading → Ranking 完整流程。

```python
from typing import List, Dict, Optional
from claw_mem.graph.multi_graph import MultiGraphMemory


class DecoupledRetriever:
    """统一检索接口 - 解耦检索管线

    Pipeline:
      Query → EngramIndex.lookup() → 种子节点
           → SpreadingActivation.activate() → 扩展节点
           → Multi-Factor Ranking → Top-K Results

    性能目标: search() < 5ms
    """

    def __init__(
        self,
        engram: "EngramIndex",
        spreader: "SpreadingActivation",
        graph: "MultiGraphMemory",
    ):
        self._engram = engram
        self._spreader = spreader
        self._graph = graph

    def search(self, query: str, top_k: int = 10,
               intent: str = "general") -> List[Dict]:
        """统一检索入口

        Args:
            query: 用户查询
            top_k: 返回结果数
            intent: 查询意图 (temporal/causal/semantic/entity/general)

        Returns:
            [{"id": str, "content": str, "score": float, "type": str}, ...]
        """
        # Phase 1: Engram O(1) 查找
        seed_results = self._engram.lookup(query, top_k=top_k * 2)
        if not seed_results:
            return self._fallback_search(query, top_k)

        # Phase 2: 扩散激活
        seed_scores = {mid: score for mid, score in seed_results}
        activations = self._spreader.activate(seed_scores, intent=intent)

        # Phase 3: 多因子排序
        return self._rank(activations, top_k)

    def _rank(self, activations: Dict[str, float],
              top_k: int) -> List[Dict]:
        """多因子排序

        因子:
          1. 激活分数 (50%)
          2. 时间新鲜度 (30%)
          3. 内容类型权重 (20%)
        """
        import time
        now = time.time()
        scored = []

        for node_id, activation in activations.items():
            node = self._graph.get_node(node_id)
            if node is None:
                continue

            content = getattr(node, 'content', '')

            # 时间新鲜度
            created_at = getattr(node, 'created_at', now)
            if hasattr(created_at, 'timestamp'):
                created_at = created_at.timestamp()
            age_days = max(0, (now - created_at) / 86400.0)
            freshness = max(0.1, 1.0 - age_days / 90.0)

            # 类型权重
            type_weights = {
                'fact': 1.0,
                'episode': 0.8,
                'reflection': 0.6,
                'concept': 0.5,
            }
            type_w = type_weights.get(
                getattr(node, 'type', None),
                0.5
            ) if hasattr(node, 'type') else 0.5

            # 综合分数
            score = activation * 0.5 + freshness * 0.3 + type_w * 0.2

            scored.append({
                "id": node_id,
                "content": content[:500],
                "score": round(score, 4),
                "type": getattr(node, 'type', None).value
                    if hasattr(node, 'type') and hasattr(node.type, 'value')
                    else str(getattr(node, 'type', '')),
                "source": "engram_spreading",
            })

        scored.sort(key=lambda x: x["score"], reverse=True)
        return scored[:top_k]

    def _fallback_search(self, query: str, top_k: int) -> List[Dict]:
        """Fallback: 回退到语义检索"""
        # 调用现有的检索方法
        return []
```

### 1.1 使用示例

```python
from claw_mem.retrieval.engram import EngramIndex
from claw_mem.retrieval.spreading import SpreadingActivation
from claw_mem.retrieval.decoupled import DecoupledRetriever
from claw_mem.graph.multi_graph import MultiGraphMemory

# 初始化
engram = EngramIndex(ngram_size=3)
graph = MultiGraphMemory()
spreader = SpreadingActivation(graph)
retriever = DecoupledRetriever(engram, spreader, graph)

# 索引
engram.index("mem_1", "用户偏好深色模式")
engram.index("mem_2", "Dark mode is preferred by the user")

# 检索
results = retriever.search("深色模式设置", top_k=5)
for r in results:
    print(f"{r['id']}: {r['score']:.3f} - {r['content'][:50]}")
```

## 2. EngramIndex API

```python
class EngramIndex:
    """O(1) N-Gram 哈希倒排索引"""

    def __init__(self, ngram_size: int = 3):
        """初始化

        Args:
            ngram_size: N-Gram 大小 (默认 3，中文适配)
        """

    def index(self, memory_id: str, content: str) -> None:
        """索引单条记忆 (O(n), n = n-gram 数量)"""

    def index_batch(self, items: List[Tuple[str, str]]) -> None:
        """批量索引 (k 条 < 10ms)"""

    def lookup(self, query: str, top_k: int = 10) -> List[Tuple[str, float]]:
        """O(1) 查找，(memory_id, jaccard_score) 列表

        Returns:
            [(memory_id, score), ...] 按分数降序
        """

    def remove(self, memory_id: str) -> None:
        """移除记忆索引"""

    def get_stats(self) -> dict:
        """获取索引统计"""

    def rebuild_from_inmemory(self, inmem_index) -> int:
        """从现有 InMemoryIndex 重建

        Returns:
            索引的记忆数量
        """
```

class SpreadingActivation:

## 6. Bridge RPC 方法

### 6.1 新增 RPC 方法

```python
# bridge.py 中新增 handlers:

"get_engram_stats": self._handle_get_engram_stats,
"rebuild_engram": self._handle_rebuild_engram,
"get_spreading_stats": self._handle_get_spreading_stats,
"get_compression_stats": self._handle_get_compression_stats,
"search_enhanced": self._handle_search_enhanced,

def _handle_get_engram_stats(self, params: Dict) -> Dict:
    return self.memory_manager.get_engram_stats()

def _handle_rebuild_engram(self, params: Dict) -> Dict:
    count = self.memory_manager.rebuild_engram()
    return {"indexed": count}

def _handle_get_spreading_stats(self, params: Dict) -> Dict:
    return self.memory_manager.get_spreading_stats()

def _handle_get_compression_stats(self, params: Dict) -> Dict:
    return self.memory_manager.get_compression_stats()

def _handle_search_enhanced(self, params: Dict) -> Dict:
    results = self.memory_manager.search(
        query=params.get("query", ""),
        limit=params.get("limit", 10),
        intent=params.get("intent", "general"),
    )
    return {"results": results}
```
