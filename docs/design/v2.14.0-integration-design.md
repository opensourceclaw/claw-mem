## 1. GroundTruthStore 设计

### 1.1 设计目标

保存原始对话的完整内容，不丢失细节，支持溯源和审计。

### 1.2 数据结构

```python
"""storage/ground_truth.py"""
import json
import os
import time
import uuid
from pathlib import Path
from typing import List, Optional, Dict
from dataclasses import dataclass, field


@dataclass
class GroundTruthRecord:
    """一条完整的原始对话记录"""
    record_id: str
    session_id: str
    messages: List[Dict]           # 完整消息列表
    timestamp: float
    metadata: Dict = field(default_factory=dict)

    def to_dict(self) -> dict:
        return {
            "record_id": self.record_id,
            "session_id": self.session_id,
            "messages": self.messages,
            "timestamp": self.timestamp,
            "metadata": self.metadata,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "GroundTruthRecord":
        return cls(**d)


class GroundTruthStore:
    """真值存储：保留原始对话的完整记录

    存储位置: ~/.claw-mem/ground_truth/
    文件格式: session_{id}.json (每个会话一个文件)

    设计原则:
      - 追加写入 (不修改已有记录)
      - 按会话组织 (一个会话一个 JSON 文件)
      - 可选的压缩 (gzip for large files)
    """

    def __init__(self, workspace: str = None):
        if workspace is None:
            workspace = os.path.expanduser("~/.claw-mem")
        self._base_dir = Path(workspace) / "ground_truth"
        self._base_dir.mkdir(parents=True, exist_ok=True)

    def _session_path(self, session_id: str) -> Path:
        """获取会话对应的文件路径"""
        safe_id = session_id.replace("/", "_").replace(":", "_")
        return self._base_dir / f"session_{safe_id}.json"

    def store_turn(self, session_id: str, messages: List[Dict],
                   metadata: Dict = None) -> str:
        """存储一轮或多轮对话

        Args:
            session_id: OpenClaw 会话 ID
            messages: 消息列表 [{"role": "user", "content": "..."}, ...]
            metadata: 额外元数据

        Returns:
            record_id
        """
        record = GroundTruthRecord(
            record_id=f"gt_{uuid.uuid4().hex[:16]}",
            session_id=session_id,
            messages=messages,
            timestamp=time.time(),
            metadata=metadata or {},
        )

        filepath = self._session_path(session_id)
        session_data = self._load_session_file(filepath)
        session_data.append(record.to_dict())
        self._save_session_file(filepath, session_data)

        return record.record_id

    def store_session(self, session_id: str,
                      all_messages: List[Dict],
                      metadata: Dict = None) -> str:
        """存储整个会话的所有消息 (agent_end 时调用)"""
        return self.store_turn(session_id, all_messages, metadata)

    def get_session(self, session_id: str) -> List[Dict]:
        """获取指定会话的全部原始对话"""
        filepath = self._session_path(session_id)
        return self._load_session_file(filepath)

    def search(self, session_id: str = None,
               keyword: str = None,
               limit: int = 50) -> List[Dict]:
        """搜索原始对话内容

        Args:
            session_id: 指定会话 (可选)
            keyword: 关键词搜索 (可选)
            limit: 最大返回数
        """
        results = []

        if session_id:
            records = self.get_session(session_id)
            for r in records:
                for m in r.get("messages", []):
                    if keyword and keyword.lower() not in str(m).lower():
                        continue
                    results.append({"session_id": session_id, "message": m})
        else:
            # 搜索所有会话
            for f in self._base_dir.glob("session_*.json"):
                sid = f.stem.replace("session_", "")
                records = self._load_session_file(f)
                for r in records:
                    for m in r.get("messages", []):
                        if keyword and keyword.lower() not in str(m).lower():
                            continue
                        results.append({"session_id": sid, "message": m})
                        if len(results) >= limit:
                            return results

        return results[:limit]

    def list_sessions(self) -> List[Dict]:
        """列出所有已存储的会话"""
        sessions = []
        for f in sorted(self._base_dir.glob("session_*.json"),
                       key=os.path.getmtime, reverse=True):
            sid = f.stem.replace("session_", "")
            sessions.append({
                "session_id": sid,
                "file_size": f.stat().st_size,
                "last_modified": f.stat().st_mtime,
            })
        return sessions

    def count_records(self) -> int:
        """统计总记录数"""
        total = 0
        for f in self._base_dir.glob("session_*.json"):
            data = self._load_session_file(f)
            total += len(data)
        return total

    # ── 内部方法 ────────────────────────────────

    def _load_session_file(self, filepath: Path) -> List[Dict]:
        """加载会话文件"""
        if not filepath.exists():
            return []
        try:
            with open(filepath, 'r', encoding='utf-8') as f:
                return json.load(f)
        except (json.JSONDecodeError, FileNotFoundError):
            return []

    def _save_session_file(self, filepath: Path, data: List[Dict]) -> None:
        """保存会话文件"""
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
```

### 1.3 与现有 Episodic Storage 的关系

```
对比:
              | Episodic Storage  | GroundTruthStore
--------------|-------------------|-------------------
存储内容      | 提炼的记忆记录    | 完整原始对话
粒度          | 单条记忆          | 全部消息
组织形式      | 按日期 .md        | 按会话 .json
数据保留      | 30天后过期        | 永久保留
查询能力      | 按日期/标签       | 按会话/关键词
与图的关系    | 节点来源          | 溯源引用
```

GroundTruthStore **不替代** Episodic Storage，而是作为补充层：
- Episodic Storage 提供快速检索的结构化记忆
- GroundTruthStore 提供完整原始对话，用于溯源和深度分析
- Fact 节点通过 `source_episode` 关联回 GroundTruth 记录

### 2.3 search() 流程修改

```python
def search(self, query, memory_type=None, limit=10,
           mode=None, include_critical=True,
           enable_graph_expansion=True,  # 新增
           graph_expansion_k: int = 10,   # 新增
           ) -> List[Dict]:
    """搜索记忆 (增强版)

    修改点:
      1. 语义检索 (现有，不变)
      2. 图扩展检索 (新增)
      3. 权重排序 (新增，考虑边衰减权重)
      4. 去重合并 (新增)
    """
    # === 第 1 步: 语义检索 (现有) ===
    raw_results = self._semantic_search(query, limit * 2, mode)

    # === 第 2 步: 图扩展检索 ===
    if (self._multi_graph and self.enable_graph
            and enable_graph_expansion):
        raw_node_ids = [r.get("id") for r in raw_results if r.get("id")]

        # 多图扩展
        expanded = self._multi_graph.multi_graph_search(
            query_embedding=self._embed(query),
            sem_nodes=raw_node_ids,
            k=graph_expansion_k,
        )

        # 加入扩展节点
        seen_ids = set(raw_node_ids)
        for node_id, score in expanded:
            if node_id not in seen_ids:
                node = self._multi_graph.get_node(node_id)
                if node:
                    raw_results.append({
                        "id": node_id,
                        "content": node.content,
                        "type": node.type.value if node.type else "",
                        "score": score,
                        "source": "graph_expansion",
                    })
                    seen_ids.add(node_id)

    # === 第 3 步: 按综合分排序 ===
    scored_results = []
    for r in raw_results:
        base_score = r.get("score", 0.5)
        # 图边权重调整 (衰减后的边权重降低相关节点分数)
        if self._multi_graph and r.get("source") == "graph_expansion":
            edge_weight = self._get_min_edge_weight(
                r.get("id"), raw_node_ids
            )
            base_score *= edge_weight  # 根据衰减边权重调整

        scored_results.append((base_score, r))

    scored_results.sort(key=lambda x: x[0], reverse=True)

    # === 第 4 步: 去重 ===
    seen_fingerprints = set()
    final_results = []
    for score, r in scored_results:
        fp = self._fingerprint(r.get("content", ""))
        if fp in seen_fingerprints:
            continue
        seen_fingerprints.add(fp)
        final_results.append(r)

    return final_results[:limit]
```

## 4. 图扩展检索算法

### 4.1 伪代码

```
function graph_expanded_search(query, k):
    # 1. 语义检索 (向量数据库)
    semantic_results = vector_search(query, top_k=k*2)
    seed_nodes = [r.node_id for r in semantic_results]

    # 2. 四子图并行扩展
    candidates = {}
    for node_id in seed_nodes:
        candidates[node_id] = 1.0  # 种子节点满分

    for subgraph in [TEMPORAL, CAUSAL, ENTITY, SEMANTIC]:
        weight = SUBGRAPH_EXPANSION_WEIGHT[subgraph]
        for seed in seed_nodes:
            neighbors = subgraph.get_neighbors(seed, max_depth=2)
            for nid, edge_weight in neighbors.items():
                score = edge_weight * weight
                candidates[nid] = max(
                    candidates.get(nid, 0), score
                )

    # 3. 加入衰减权重
    for nid in candidates:
        edge_weight = get_min_decay_weight_to_seeds(nid, seed_nodes)
        candidates[nid] *= edge_weight

    # 4. 排序去重
    sorted_results = sorted(candidates.items(),
                           key=lambda x: x[1], reverse=True)
    return sorted_results[:k]
```

### 4.2 扩展权重配置

| 子图 | 扩展权重 | 深度 | 说明 |
|------|:------:|:----:|------|
| TEMPORAL | 0.6 | 2 | 当前任务的前后相关节点 |
| CAUSAL | 0.5 | 1 | 因果链条 (原因/结果) |
| SEMANTIC | 0.4 | 1 | 语义相似的相邻节点 |
| ENTITY | 0.3 | 1 | 共享实体的节点 |

## 5. 错误处理

### 5.1 故障模式与恢复策略

| 故障模式 | 影响 | 恢复策略 |
|----------|------|----------|
| **图索引更新失败** | 部分边未创建 | 重试队列 (最多 3 次)，检索时回退到纯语义检索 |
| **衰减计算失败** | 权重未更新 | 下一次调度周期自动重试 |
| **边构建 OOM** | 图索引内存溢出 | 限制 BFS 深度和扩展节点数，LRU 淘汰旧节点 |
| **JSON 序列化失败** | 图结构无法持久化 | 记录错误日志，下次 agent_end 重试 |
| **GroundTruth 写入失败** | 原始对话丢失 | 不影响主流程，记录警告日志 |
| **图数据不一致** | 边指向不存在的节点 | 定期一致性检查 (每周)，自动修复孤边 |
| **衰减任务死锁** | 调度器卡死 | 超时机制 (30s)，自动重启调度器 |

### 5.3 一致性检查

```python
def verify_graph_consistency(graph: MultiGraphMemory) -> List[str]:
    """检查图数据一致性

    Returns:
        检测到的问题列表
    """
    issues = []

    for sg_type, subgraph in graph._graphs.items():
        # 检查: 所有边中的节点是否都在 _node_index 中
        for (s, t) in subgraph.edge_weights.keys():
            if s not in graph._node_index:
                issues.append(f"Missing source node: {s}")
            if t not in graph._node_index:
                issues.append(f"Missing target node: {t}")

    # 检查: 邻接表与 edge_weights 是否一致
    for sg_type, subgraph in graph._graphs.items():
        for node_id, neighbors in subgraph.adjacency.items():
            for neighbor_id, weight in neighbors:
                key = (node_id, neighbor_id)
                if key not in subgraph.edge_weights:
                    issues.append(
                        f"Adjacency/weights mismatch: {node_id} → {neighbor_id}"
                    )

    return issues
```

## 6. 性能优化

### 6.1 图索引内存优化

| 优化策略 | 实现 | 效果 |
|----------|------|------|
| **节点限制** | 最多保留 10,000 个节点 | 内存 < 5MB |
| **边限制** | 每个子图最多 50,000 条边 | 内存 < 2MB |
| **LRU 淘汰** | 最久未访问的节点优先淘汰 | 自动回收 |
| **字符串驻留** | intern(str) 减少重复字符串 | 减少 30% 内存 |
| **稀疏邻接表** | 仅在度数 > 0 时才创建 List | 减少空列表开销 |

```python
class BoundedMultiGraphMemory(MultiGraphMemory):
    """带有节点数量限制的图"""

    def __init__(self, max_nodes: int = 10000):
        super().__init__()
        self._max_nodes = max_nodes
        self._access_times: Dict[str, float] = {}

    def add_node(self, memory_id, content, node_type, **metadata):
        if len(self._node_index) >= self._max_nodes:
            self._evict_lru()
        super().add_node(memory_id, content, node_type, **metadata)

    def _evict_lru(self):
        """淘汰最久未访问的节点"""
        if not self._access_times:
            return
        oldest = min(self._access_times, key=self._access_times.get)
        if oldest in self._node_index:
            del self._node_index[oldest]
            for g in self._graphs.values():
                g.nodes.discard(oldest)
        del self._access_times[oldest]
```

### 6.2 检索剪枝策略

```python
def graph_expansion_with_pruning(seed_nodes, subgraph, max_depth, max_branches):
    """带剪枝的图扩展"""
    visited = dict(seed_nodes)  # node_id → score

    current = list(seed_nodes.items())
    for depth in range(max_depth):
        if not current:
            break
        next_layer = []
        for node_id, parent_score in current:
            neighbors = subgraph.get_edges_from(node_id)
            # 剪枝 1: 只取高权重邻居
            sorted_neighbors = sorted(
                neighbors, key=lambda x: x[1], reverse=True
            )[:max_branches]

            for neighbor_id, edge_weight in sorted_neighbors:
                if neighbor_id in visited:
                    continue
                score = parent_score * edge_weight * (0.8 ** depth)
                # 剪枝 2: 低分跳过
                if score < 0.1:
                    continue
                visited[neighbor_id] = score
                next_layer.append((neighbor_id, score))
        current = next_layer

    return visited
```

### 6.3 批量操作

```python
def batch_add_conversation(mg: MultiGraphMemory,
                           turns: List[Dict],
                           session_id: str) -> List[str]:
    """批量添加一个完整对话"""
    node_ids = []

    # 批量注册节点
    prev_id = None
    for turn in turns:
        if turn.get("role") == "user":
            node_id = f"mem_{uuid.uuid4().hex[:12]}"
            mg.add_node(node_id, turn["content"],
                       NodeType.EPISODE, session_id=session_id)
            node_ids.append(node_id)

            # 批量建边 (事务性)
            if prev_id:
                mg.add_edge(prev_id, node_id, EdgeType.NEXT, weight=1.0)
            prev_id = node_id

    return node_ids
```
