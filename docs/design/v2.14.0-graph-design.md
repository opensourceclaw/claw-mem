## 1. 数据结构

### 1.1 复用现有节点类型 (`graph/nodes.py`)

v2.14.0 的图结构基于现有的 `graph/` 模块节点和边类型，不做破坏性修改。

```python
# 现有 enum (保持不变)
class NodeType(Enum):
    EPISODE = "episode"
    FACT = "fact"
    REFLECTION = "reflection"
    CONCEPT = "concept"

class EdgeType(Enum):
    NEXT = "next"                    # 有向: Episode → Episode (时序)
    DERIVED_FROM = "derived_from"    # 有向: Fact → Episode (因果)
    SYNTHESIZED_FROM = "synthesized_from"  # 有向: Reflection → Episode/Fact
    RELATED_TO = "related_to"        # 无向: 任意 ↔ 任意 (语义关联)
    HAS_CONCEPT = "has_concept"      # 有向: 任意 → Concept (实体归属)
```

### 1.2 新增：子图索引结构

```python
from dataclasses import dataclass, field
from typing import Dict, List, Set, Tuple, Optional
from enum import Enum
from collections import defaultdict


class SubGraphType(Enum):
    """四正交子图类型"""
    SEMANTIC = "semantic"   # 语义相似性图
    TEMPORAL = "temporal"   # 时序图
    CAUSAL = "causal"       # 因果图
    ENTITY = "entity"       # 实体共现图


@dataclass
class GraphEdge:
    """轻量级图边，用于子图内部索引"""
    source_id: str
    target_id: str
    weight: float          # 0.0 ~ 1.0, 初始值 1.0
    edge_type: str
    created_at: float      # timestamp

    def to_dict(self) -> dict:
        return {
            "s": self.source_id,
            "t": self.target_id,
            "w": round(self.weight, 4),
            "e": self.edge_type,
            "c": self.created_at,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "GraphEdge":
        return cls(
            source_id=d["s"],
            target_id=d["t"],
            weight=d.get("w", 1.0),
            edge_type=d.get("e", ""),
            created_at=d.get("c", 0.0),
        )


@dataclass
class SubGraph:
    """单个子图：邻接表 + 逆邻接表实现"""
    name: SubGraphType
    # 邻接表: node_id → [(neighbor_id, weight), ...]
    adjacency: Dict[str, List[Tuple[str, float]]] = field(default_factory=dict)
    # 逆邻接表 (有向图反向查询): node_id → [(neighbor_id, weight), ...]
    reverse_adjacency: Dict[str, List[Tuple[str, float]]] = field(default_factory=dict)
    # 边权重映射: (source, target) → weight (O(1) 权重查找)
    edge_weights: Dict[Tuple[str, str], float] = field(default_factory=dict)
    # 节点集合 (快速存在性检查)
    nodes: Set[str] = field(default_factory=set)
    # 统计
    edge_count: int = 0

    def add_node(self, node_id: str) -> None:
        """注册节点到子图"""
        if node_id not in self.nodes:
            self.nodes.add(node_id)
            self.adjacency.setdefault(node_id, [])
            self.reverse_adjacency.setdefault(node_id, [])

    def add_edge(self, source: str, target: str, weight: float = 1.0,
                 directed: bool = True) -> None:
        """添加边到子图"""
        self.add_node(source)
        self.add_node(target)

        # 正邻接
        self.adjacency[source].append((target, weight))
        if directed:
            # 逆邻接 (有向边)
            self.reverse_adjacency[target].append((source, weight))
        else:
            # 无向边: 双向添加
            self.adjacency[target].append((source, weight))

        self.edge_weights[(source, target)] = weight
        self.edge_count += 1

    def get_neighbors(self, node_id: str,
                      max_depth: int = 1) -> Dict[str, float]:
        """获取节点的邻居 (BFS 广度优先)

        Args:
            node_id: 起始节点
            max_depth: 最大遍历深度

        Returns:
            {neighbor_id: accumulated_weight}
        """
        if node_id not in self.nodes:
            return {}

        visited: Dict[str, float] = {}
        queue = [(node_id, 1.0)]  # (node_id, path_weight)

        for _ in range(max_depth + 1):
            if not queue:
                break
            next_queue = []
            for current, path_weight in queue:
                for neighbor, weight in self.adjacency.get(current, []):
                    if neighbor not in visited and neighbor != node_id:
                        new_weight = path_weight * weight
                        visited[neighbor] = max(
                            visited.get(neighbor, 0), new_weight
                        )
                        next_queue.append((neighbor, new_weight))
            queue = next_queue

        return visited

    def get_edges_from(self, node_id: str) -> List[Tuple[str, float]]:
        """获取从 node_id 出发的所有边"""
        return self.adjacency.get(node_id, [])

    def get_edges_to(self, node_id: str) -> List[Tuple[str, float]]:
        """获取指向 node_id 的所有边"""
        return self.reverse_adjacency.get(node_id, [])

    def has_edge(self, source: str, target: str) -> bool:
        """检查边是否存在"""
        return (source, target) in self.edge_weights

    def update_weight(self, source: str, target: str, weight: float) -> bool:
        """更新边权重"""
        key = (source, target)
        if key in self.edge_weights:
            self.edge_weights[key] = weight
            return True
        return False

    def to_dict(self) -> dict:
        """序列化为字典（用于 JSON 持久化）"""
        edges = []
        for (s, t), w in self.edge_weights.items():
            edges.append(GraphEdge(
                source_id=s, target_id=t, weight=w,
                edge_type=self.name.value, created_at=0.0,
            ).to_dict())
        return {
            "name": self.name.value,
            "edge_count": self.edge_count,
            "node_count": len(self.nodes),
            "edges": edges,
        }

    @classmethod
    def from_dict(cls, d: dict) -> "SubGraph":
        """从字典反序列化"""
        g = cls(name=SubGraphType(d["name"]))
        for e in d.get("edges", []):
            edge = GraphEdge.from_dict(e)
            g.add_edge(edge.source_id, edge.target_id, edge.weight,
                       directed=(g.name != SubGraphType.SEMANTIC))
        return g

    @property
    def memory_estimate(self) -> int:
        """估算内存占用 (bytes)"""
        # 每个节点 ~100 bytes (set entry + list overhead)
        # 每条边 ~80 bytes (tuple + list entry)
        return len(self.nodes) * 100 + self.edge_count * 80
```

### 1.3 边类型到子图的映射

| `NEXT` | TEMPORAL | 有向 | 时间接近度 |
| `DERIVED_FROM` | CAUSAL | 有向 | 因果强度 |

| `RELATED_TO` | SEMANTIC | 无向 | 语义相似度 |
| `HAS_CONCEPT` | ENTITY | 有向 | 实体关联强度 |

## 2. 四正交图 (MultiGraphMemory)

### 2.1 类定义

```python
from threading import Lock
from typing import List, Set, Dict, Tuple, Optional
from claw_mem.graph.nodes import Node, NodeType
from claw_mem.graph.edges import Edge, EdgeType


class MultiGraphMemory:
    """四正交图索引层：管理四个独立的子图视图

    四个正交维度：
      - SEMANTIC: 语义相似性关系
      - TEMPORAL: 时间先后关系
      - CAUSAL: 因果推导关系
      - ENTITY: 实体共现关系

    同一对节点可以在多个子图中同时存在边。
    """

    def __init__(self):
        self._graphs: Dict[SubGraphType, SubGraph] = {
            SubGraphType.SEMANTIC: SubGraph(SubGraphType.SEMANTIC),
            SubGraphType.TEMPORAL: SubGraph(SubGraphType.TEMPORAL),
            SubGraphType.CAUSAL: SubGraph(SubGraphType.CAUSAL),
            SubGraphType.ENTITY: SubGraph(SubGraphType.ENTITY),
        }
        self._node_index: Dict[str, Node] = {}  # node_id → Node
        self._lock = Lock()

    # ── 节点管理 ────────────────────────────────────────

    def add_node(self, memory_id: str, content: str,
                 node_type: NodeType, **metadata) -> None:
        """注册一个记忆节点到图中"""
        with self._lock:
            if memory_id in self._node_index:
                return  # 幂等
            node = create_node(node_type, content, id=memory_id, **metadata)
            self._node_index[memory_id] = node
            for g in self._graphs.values():
                g.add_node(memory_id)

    def get_node(self, memory_id: str) -> Optional[Node]:
        return self._node_index.get(memory_id)

    # ── 边管理 ────────────────────────────────────────

    def add_edge(self, source_id: str, target_id: str,
                 edge_type: EdgeType, weight: float = 1.0) -> None:
        """根据边类型自动路由到对应的子图"""
        subgraph = self._map_edge_to_subgraph(edge_type)
        with self._lock:
            self._graphs[subgraph].add_edge(
                source_id, target_id, weight,
                directed=edge_type != EdgeType.RELATED_TO
            )

    @staticmethod
    def _map_edge_to_subgraph(edge_type: EdgeType) -> SubGraphType:
        return {
            EdgeType.NEXT: SubGraphType.TEMPORAL,
            EdgeType.DERIVED_FROM: SubGraphType.CAUSAL,
            EdgeType.SYNTHESIZED_FROM: SubGraphType.CAUSAL,
            EdgeType.RELATED_TO: SubGraphType.SEMANTIC,
            EdgeType.HAS_CONCEPT: SubGraphType.ENTITY,
        }[edge_type]

    # ── 检索 ────────────────────────────────────────

    def get_related(self, memory_id: str, subgraph: SubGraphType,
                    limit: int = 10) -> List[str]:
        """在指定子图中获取相关节点"""
        with self._lock:
            neighbors = self._graphs[subgraph].get_neighbors(memory_id)
        # 按权重排序，取 top-k
        sorted_neighbors = sorted(
            neighbors.items(), key=lambda x: x[1], reverse=True
        )
        return [nid for nid, _ in sorted_neighbors[:limit]]

    def get_expanded_nodes(self, node_ids: List[str],
                           subgraphs: List[SubGraphType] = None,
                           max_depth: int = 1,
                           max_expansion: int = 50) -> Dict[str, float]:
        """从多个种子节点出发进行图扩展

        Args:
            node_ids: 种子节点 ID 列表
            subgraphs: 要查询的子图，默认全部
            max_depth: BFS 最大深度
            max_expansion: 最大扩展节点数

        Returns:
            {node_id: aggregated_weight}
        """
        if subgraphs is None:
            subgraphs = list(SubGraphType)

        all_nodes: Dict[str, float] = {}
        for nid in node_ids:
            all_nodes[nid] = 1.0  # 种子节点权重最高

        with self._lock:
            for subgraph in subgraphs:
                for seed in node_ids:
                    neighbors = self._graphs[subgraph].get_neighbors(
                        seed, max_depth=max_depth
                    )
                    for nid, weight in neighbors.items():
                        if nid not in all_nodes:
                            all_nodes[nid] = weight
                        else:
                            all_nodes[nid] += weight  # 多子图聚合

        # 按权�排序，限制数量
        expanded = sorted(
            all_nodes.items(), key=lambda x: x[1], reverse=True
        )
        return dict(expanded[:max_expansion])

    def multi_graph_search(self, query_embedding: List[float],
                           sem_nodes: List[str],
                           k: int = 10) -> List[Tuple[str, float]]:
        """多图联合检索

        策略:
          1. 语义子图: 找到语义相似节点
          2. 时序子图: 跟随 NEXT 链条
          3. 因果子图: 跟随 DERIVED_FROM 链条
          4. 实体子图: 实体共现扩展
          5. 加权合并结果

        Args:
            query_embedding: 查询的向量表示
            sem_nodes: 初始语义检索得到的节点 ID
            k: 返回数量

        Returns:
            [(node_id, combined_score), ...]
        """
        all_candidates: Dict[str, float] = {}

        # Phase 1: 语义种子 (权重 1.0)
        for nid in sem_nodes:
            all_candidates[nid] = 1.0

        # Phase 2: 图扩展 (四个子图，权重由子图重要性决定)
        expansion_weights = {
            SubGraphType.SEMANTIC: 0.8,
            SubGraphType.TEMPORAL: 0.6,
            SubGraphType.CAUSAL: 0.5,
            SubGraphType.ENTITY: 0.3,
        }

        for subgraph, ew in expansion_weights.items():
            neighbors = self.get_expanded_nodes(
                sem_nodes, subgraphs=[subgraph], max_depth=1, max_expansion=30
            )
            for nid, w in neighbors.items():
                if nid not in all_candidates:
                    all_candidates[nid] = w * ew
                else:
                    all_candidates[nid] += w * ew

        # 排序取 top-k
        sorted_c = sorted(
            all_candidates.items(), key=lambda x: x[1], reverse=True
        )
        return sorted_c[:k]

    # ── 衰减 ────────────────────────────────────────

    def apply_decay(self, edge_weights: Dict[Tuple[str, str], float]) -> int:
        """批量应用边权重衰减

        Args:
            edge_weights: {(source_id, target_id): new_weight}

        Returns:
            成功更新的边数
        """
        updated = 0
        with self._lock:
            for (s, t), weight in edge_weights.items():
                for g in self._graphs.values():
                    if g.update_weight(s, t, weight):
                        updated += 1
        return updated

    def remove_expired_edges(self, threshold: float = 0.05) -> int:
        """移除权重低于阈值的边"""
        removed = 0
        with self._lock:
            for g in self._graphs.values():
                expired = [
                    (s, t) for (s, t), w in g.edge_weights.items()
                    if w < threshold
                ]
                for (s, t) in expired:
                    del g.edge_weights[(s, t)]
                    # 从邻接表移除
                    g.adjacency[s] = [
                        (n, w) for n, w in g.adjacency[s] if n != t
                    ]
                    g.reverse_adjacency[t] = [
                        (n, w) for n, w in g.reverse_adjacency[t] if n != s
                    ]
                    removed += 1
                g.edge_count = len(g.edge_weights)
        return removed

    # ── 统计 ────────────────────────────────────────

    def get_stats(self) -> dict:
        with self._lock:
            return {
                "total_nodes": len(self._node_index),
                "subgraphs": {
                    g.name.value: {
                        "nodes": len(g.nodes),
                        "edges": g.edge_count,
                        "memory_bytes": g.memory_estimate,
                    }
                    for g in self._graphs.values()
                },
            }

    # ── 持久化 ────────────────────────────────────────

    def to_dict(self) -> dict:
        """序列化为可持久化的字典"""
        with self._lock:
            return {
                "nodes": {
                    nid: node.to_dict()
                    for nid, node in self._node_index.items()
                },
                "subgraphs": {
                    g.name.value: g.to_dict()
                    for g in self._graphs.values()
                },
            }

    @classmethod
    def from_dict(cls, d: dict) -> "MultiGraphMemory":
        mg = cls()
        for nid, nd in d.get("nodes", {}).items():
            mg._node_index[nid] = Node.from_dict(nd)
        for name, sd in d.get("subgraphs", {}).items():
            sg_type = SubGraphType(name)
            mg._graphs[sg_type] = SubGraph.from_dict(sd)
        return mg
```

### 2.2 图构建流程

```
MemoryManager.store(text, type="episodic")
         │
         ▼
  Node 创建 (EpisodeNode / FactNode / ConceptNode)
         │
         ├──→ MultiGraphMemory.add_node(node_id, content, node_type)
         │
         ├──→ 边构建 (异步):
         │      │
         │      ├── TEMPORAL: 与最近 Episode 建立 NEXT 边
         │      ├── SEMANTIC: 计算与现有节点的语义相似度
         │      │             高于阈值则建立 RELATED_TO 边
         │      ├── CAUSAL:   如果节点有 source_episode_id
         │      │             建立 DERIVED_FROM 边
         │      └── ENTITY:   提取实体 → 建立 HAS_CONCEPT 边
         │
         └──→ 持久化 (异步): 图结构序列化为 JSON
```

### 2.3 边构建阈值

| 边类型 | 条件 | 阈值 |
|--------|------|:----:|
| NEXT | 同 session 内最近 Episode | time_gap < 5min |
| RELATED_TO | embedding 余弦相似度 | > 0.7 |
| DERIVED_FROM | Fact/Reflection 显式关联 | 必须满足 |
| HAS_CONCEPT | 实体提取匹配 | 必须满足 |

### 3.2 DualLayerMemory 类

```python
import uuid
from threading import Lock
from typing import List, Set, Dict, Optional


class DualLayerMemory:
    """双层记忆结构

    Layer 1 - Event Progression Graph (事件进展图):
      - 将记忆节点聚类为"事件"
      - 事件之间有时序关系 (before/after)
      - 支持事件链式回溯

    Layer 2 - Topic Associative Network (主题关联网络):
      - 将事件和节点按语义聚�为"主题"
      - 主题之间有语义关联权重
      - 支持跨会话主题导航
    """

    def __init__(self):
        self._events: Dict[str, Event] = {}
        self._topics: Dict[str, Topic] = {}
        self._event_chain: Dict[str, List[str]] = {}  # event_id → [prev_event_ids]
        self._topic_links: Dict[Tuple[str, str], float] = {}  # (t1, t2) → weight
        self._lock = Lock()

    # ── Layer 1: Event Progression Graph ─────────────────

    def add_event(self, title: str, description: str = "",
                  node_ids: List[str] = None,
                  session_id: str = None,
                  tags: List[str] = None) -> str:
        """创建新事件"""
        event_id = f"evt_{uuid.uuid4().hex[:12]}"
        now = datetime.now().timestamp()
        event = Event(
            event_id=event_id,
            title=title,
            description=description,
            node_ids=node_ids or [],
            session_id=session_id,
            start_time=now,
            tags=tags or [],
        )
        with self._lock:
            self._events[event_id] = event

            # 自动建立时序链: 找到最近的同一 session 事件
            if session_id:
                prev = self._find_latest_event_in_session(session_id)
                if prev:
                    self._event_chain.setdefault(event_id, []).append(prev)

        return event_id

    def _find_latest_event_in_session(self, session_id: str) -> Optional[str]:
        """找到指定 session 的最新事件"""
        best_time = 0.0
        best_id = None
        for eid, evt in self._events.items():
            if evt.session_id == session_id and (
                evt.end_time or evt.start_time
            ) > best_time:
                best_time = evt.end_time or evt.start_time
                best_id = eid
        return best_id

    def link_events(self, event1_id: str, event2_id: str) -> None:
        """显式关联两个事件（双向）"""
        with self._lock:
            self._event_chain.setdefault(event1_id, []).append(event2_id)
            self._event_chain.setdefault(event2_id, []).append(event1_id)

    def get_event_chain(self, event_id: str) -> List[Event]:
        """获取事件链（前驱事件序列）"""
        visited = set()
        result = []
        queue = [event_id]

        while queue:
            eid = queue.pop(0)
            if eid in visited or eid not in self._events:
                continue
            visited.add(eid)
            result.append(self._events[eid])
            queue.extend(self._event_chain.get(eid, []))

        return result

    def find_events_by_tags(self, tags: List[str]) -> List[Event]:
        """按标签查找事件"""
        result = []
        for evt in self._events.values():
            if any(t in evt.tags for t in tags):
                result.append(evt)
        return result

    # ── Layer 2: Topic Associative Network ─────────────────

    def add_topic(self, name: str, description: str = "",
                  node_ids: List[str] = None,
                  event_ids: List[str] = None,
                  keywords: List[str] = None) -> str:
        """创建新主题"""
        topic_id = f"tpc_{uuid.uuid4().hex[:12]}"
        now = datetime.now().timestamp()
        topic = Topic(
            topic_id=topic_id,
            name=name,
            description=description,
            node_ids=node_ids or [],
            event_ids=event_ids or [],
            keywords=keywords or [],
            created_at=now,
            updated_at=now,
        )
        with self._lock:
            self._topics[topic_id] = topic
        return topic_id

    def link_topics(self, topic1_id: str, topic2_id: str,
                    weight: float = 0.5) -> None:
        """关联两个主题"""
        with self._lock:
            self._topic_links[(topic1_id, topic2_id)] = weight
            self._topic_links[(topic2_id, topic1_id)] = weight  # 对称

    def get_related_topics(self, topic_id: str,
                           min_weight: float = 0.3) -> List[Tuple[Topic, float]]:
        """获取相关主题"""
        results = []
        with self._lock:
            for (t1, t2), w in self._topic_links.items():
                if t1 == topic_id and w >= min_weight and t2 in self._topics:
                    results.append((self._topics[t2], w))
        return sorted(results, key=lambda x: x[1], reverse=True)

    def search_by_keywords(self, keywords: List[str]) -> List[Topic]:
        """按关键词搜索主题"""
        results = []
        for topic in self._topics.values():
            score = len(set(keywords) & set(topic.keywords))
            if score > 0:
                results.append((topic, score))
        return [t for t, _ in sorted(results, key=lambda x: x[1], reverse=True)]

    # ── 持久化 ────────────────────────────────────────

    def to_dict(self) -> dict:
        return {
            "events": {eid: evt.to_dict() for eid, evt in self._events.items()},
            "topics": {tid: tpc.to_dict() for tid, tpc in self._topics.items()},
            "event_chain": self._event_chain,
            "topic_links": {
                f"{k[0]}||{k[1]}": v
                for k, v in self._topic_links.items()
            },
        }

    @classmethod
    def from_dict(cls, d: dict) -> "DualLayerMemory":
        dm = cls()
        dm._events = {
            eid: Event.from_dict(ed) for eid, ed in d.get("events", {}).items()
        }
        dm._topics = {
            tid: Topic.from_dict(td) for tid, td in d.get("topics", {}).items()
        }
        dm._event_chain = d.get("event_chain", {})
        dm._topic_links = {}
        for k, v in d.get("topic_links", {}).items():
            t1, t2 = k.split("||", 1)
            dm._topic_links[(t1, t2)] = v
        return dm
```

### 3.3 双层结构与四正交图的关系

```
┌─────────────────────────────────────────────────┐
│            DualLayerMemory (双层)                │
│                                                 │
│  Layer 1: Event Progression Graph               │
│           event_1 → event_2 → event_3           │
│              ↓         ↓         ↓              │
│  Layer 2: Topic Associative Network             │
│           topic_A ←→ topic_B                    │
│              ↓  ↘      ↓                        │
├─────────────────────────────────────────────────┤
│           MultiGraphMemory (四正交)              │
│                                                 │
│  [Nodes: episode/fact/reflection/concept]       │
│  [Edges: NEXT | RELATED_TO | DERIVED_FROM | ...]│
└─────────────────────────────────────────────────┘

映射关系:
  - 1 Event = 多个 Node (同一 session/主题)
  - 1 Topic = 多个 Event + 多个 Node
  - Event 之间: 对应 TEMPORAL 子图的 NEXT 链条
  - Topic 之间: 在 SEMANTIC 子图上进一步聚合
```

### 5.3 实体倒排索引

```python
class EntityIndex:
    """实体 → 节点的倒排索引"""
    def __init__(self):
        self._index: Dict[str, Set[str]] = defaultdict(set)

    def index_node(self, node_id: str, entities: List[str]) -> None:
        for entity in entities:
            self._index[entity.lower()].add(node_id)

    def get_nodes_by_entity(self, entity: str) -> Set[str]:
        return self._index.get(entity.lower(), set())

    def get_co_occurring(self, entity: str) -> List[Tuple[str, int]]:
        """找到与 entity 共现的其他实体，按共现次数排序"""
        co_entities: Dict[str, int] = defaultdict(int)
        for nid in self.get_nodes_by_entity(entity):
            for other, nodes in self._index.items():
                if other != entity and nid in nodes:
                    co_entities[other] += 1
        return sorted(co_entities.items(), key=lambda x: x[1], reverse=True)
```

## 6. 核心算法

### 6.1 图构建算法 (异步)

```python
async def build_graph_edges_async(mg: MultiGraphMemory,
                                   new_node_id: str,
                                   new_node: Node,
                                   session_nodes: List[str],
                                   embedder) -> None:
    """异步构建新节点的所有边

    必须在 MemoryManager.store() 之后立即调度，
    但不阻塞 store() 的返回。
    """
    import asyncio

    await asyncio.to_thread(_build_edges_sync, mg, new_node_id,
                            new_node, session_nodes, embedder)


def _build_edges_sync(mg: MultiGraphMemory,
                      new_node_id: str,
                      new_node: Node,
                      session_nodes: List[str],
                      embedder) -> None:
    # 1. TEMPORAL: NEXT 边 (同 session 内最近节点)
    if session_nodes and hasattr(new_node, 'session_id'):
        recent = session_nodes[-2] if len(session_nodes) >= 2 else None
        if recent and recent != new_node_id:
            time_diff = (getattr(new_node, 'created_at', 0) -
                        getattr(mg.get_node(recent), 'created_at', 0))
            if time_diff < 300:  # 5 分钟内
                mg.add_edge(recent, new_node_id, EdgeType.NEXT,
                            weight=1.0 - min(time_diff / 600, 0.5))

    # 2. SEMANTIC: RELATED_TO 边 (语义相似节点)
    if embedder and hasattr(new_node, 'content'):
        embedding = embedder.embed(new_node.content)
        for nid, node in mg._node_index.items():
            if nid == new_node_id:
                continue
            if hasattr(node, 'embedding') and node.embedding:
                sim = cosine_similarity(embedding, node.embedding)
                if sim > 0.7:
                    mg.add_edge(new_node_id, nid, EdgeType.RELATED_TO,
                                weight=sim)

    # 3. CAUSAL: DERIVED_FROM 边 (Fact/Reflection)
    if hasattr(new_node, 'source_episode'):
        source_id = new_node.source_episode
        if source_id in mg._node_index:
            mg.add_edge(new_node_id, source_id, EdgeType.DERIVED_FROM)

    # 4. ENTITY: HAS_CONCEPT 边 (实体提取)
    entities = _extract_entities(new_node.content)  # 分词 + NER
    for entity in entities:
        concept_id = f"concept_{hash(entity) % 10**12}"
        mg.add_node(concept_id, entity, NodeType.CONCEPT)
        mg.add_edge(new_node_id, concept_id, EdgeType.HAS_CONCEPT)
```

### 6.2 图扩展检索算法

```
输入: query_embedding, k=10
输出: ranked_node_ids  # 按综合分数排序

步骤:
  1. 语义检索: 在向量数据库中检索 top-k * 2 = 20 个候选节点
     评分: sim_score = cosine(query_emb, node_emb)

  2. 图扩展: 对每个候选节点:
     a. TEMPORAL子图: BFS depth=1, 获取前后节点
        score += 0.6 * edge_weight * sim_score
     b. CAUSAL子图: BFS depth=1, 获取原因/结果节点
        score += 0.5 * edge_weight * sim_score
     c. ENTITY子图: 获取同实体节点
        score += 0.3 * co_occurrence_score * sim_score
     d. SEMANTIC子图: 获取语义相关节点
        score += 0.4 * edge_weight * sim_score

  3. 排序: 所有候选节点按综合分数降序
  4. 截断: 取 top-k
  5. 去重: 对相同 content 的节点去重
```

### 6.3 多图聚合评分公式

```
对于节点 n:
  combined_score(n) = Σ (w_subgraph * gb_score(n, subgraph))

其中:
  w_subgraph: 子图权重
    - SEMANTIC: 0.8
    - TEMPORAL: 0.6
    - CAUSAL: 0.5
    - ENTITY: 0.3

  gb_score(n, subgraph): 图传播分数
    = Σ (seed_score * Π edge_weights_on_path)  # 所有从种子到n的路径
```
